Use this endpoint to fetch the source data for a project associated with your particular domain.

This requires a valid projectId as well as the signature parameters to authenticate the call.

**\[POST\]** [https://pitchprint.net/api/runtime/fetch-project-source](https://pitchprint.net/api/runtime/fetch-project-source)

##### BODY PARAMS

&lt;table class="table-striped table-bordered">&lt;tbody>&lt;tr>&lt;td class="required">apiKey&lt;/td>&lt;td>Your domain's api key&lt;/td>&lt;/tr>&lt;tr>&lt;td class="required">timestamp&lt;/td>&lt;td>Timestamp generated at runtime&lt;/td>&lt;/tr>&lt;tr>&lt;td class="required">signature&lt;/td>&lt;td>Signature = MD5(apikey + secretKey + timestamp)&lt;/td>&lt;/tr>&lt;tr>&lt;td class="required">projectId&lt;/td>&lt;td>ID of a project&lt;/td>&lt;/tr>&lt;/tbody>&lt;/table>

* * *

##### SAMPLE CODE

&lt;?php

define('PITCH\_APIKEY', 'your-api-key');
define('PITCH\_SECRETKEY', 'your-secret-key');

function generateSignature () &#123;
    $timestamp = time();
    $signature = md5(PITCH\_APIKEY . PITCH\_SECRETKEY . $timestamp);
    return array ('timestamp'=>$timestamp, 'apiKey'=>PITCH\_APIKEY, 'signature'=>$signature);
&#125;

$ch = curl\_init();
curl\_setopt($ch, CURLOPT\_URL, 'https://pitchprint.net/api/runtime/fetch-project-source');
curl\_setopt($ch, CURLOPT\_POST, true);

$opts = generateSignature();
$opts\['projectId'\] = 'your-project-id-here';

curl\_setopt($ch, CURLOPT\_POSTFIELDS, http\_build\_query($opts));
curl\_setopt($ch, CURLOPT\_RETURNTRANSFER, true);

$output = curl\_exec($ch);
$http\_status = curl\_getinfo($ch, CURLINFO\_HTTP\_CODE);
$content\_type = curl\_getinfo($ch, CURLINFO\_CONTENT\_TYPE);
$curlerr = curl\_error($ch);
curl\_close($ch);

header('Content-Type: application/json');

echo $output;

?>
